1. program → program id ( identifier_list ) ; declarations subprogram_declarations compound_statement

FIRST(program) = { program }

FOLLOW(program) = { $ }



2. identifier_list → id | identifier_list , id

FIRST(identifier_list) = { id }

FOLLOW(identifier_list) = { ), :, , }

---

3. declarations → declarations var identifier_list : type ; | null

FIRST(declarations) = { var, ε }

FOLLOW(declarations) = { function, procedure, begin }

---

4. type → standard_type | array [ num .. num ] of standard_type

FIRST(type) = { integer, real, array }

FOLLOW(type) = { ; }

---

5. standard_type → integer | real

FIRST(standard_type) = { integer, real }

FOLLOW(standard_type) = { ; }

---

6. subprogram_declarations → subprogram_declarations subprogram_declaration ; | null

FIRST(subprogram_declarations) = { function, procedure, ε }

FOLLOW(subprogram_declarations) = { begin }

---

7. subprogram_declaration → subprogram_head declarations compound_statement

FIRST(subprogram_declaration) = { function, procedure }

FOLLOW(subprogram_declaration) = { ;, begin }

---

8. subprogram_head → function id arguments : standard_type ; | procedure id arguments ;

FIRST(subprogram_head) = { function, procedure }

FOLLOW(subprogram_head) = { var, begin }

---

9. arguments → ( parameter_list ) | null

FIRST(arguments) = { (, ε }

FOLLOW(arguments) = { :, ; }

---

10. parameter_list → identifier_list : type | parameter_list ; identifier_list : type

FIRST(parameter_list) = { id }

FOLLOW(parameter_list) = { ) }

---

11. compound_statement → begin optional_statements end

FIRST(compound_statement) = { begin }

FOLLOW(compound_statement) = { $, ;, else, end }

---

12. optional_statements → statement_list | null

FIRST(optional_statements) = { id, begin, if, while, ε }

FOLLOW(optional_statements) = { end }

---

13. statement_list → statement | statement_list ; statement

FIRST(statement_list) = { id, begin, if, while }

FOLLOW(statement_list) = { end }

---

14. statement → variable assignop expression  
              | procedure_statement  
              | compound_statement  
              | if expression then statement else statement  
              | while expression do statement

FIRST(statement) = { id, begin, if, while }

FOLLOW(statement) = { ;, else, end }

---

15. variable → id | id [ expression ]

FIRST(variable) = { id }

FOLLOW(variable) = { assignop }

---

16. procedure_statement → id | id ( expression_list )

FIRST(procedure_statement) = { id }

FOLLOW(procedure_statement) = { ;, else, end }

---

17. expression_list → expression | expression_list , expression

FIRST(expression_list) = { id, num, (, not, +, - }

FOLLOW(expression_list) = { ), ] }

---

18. expression → simple_expression | simple_expression relop simple_expression

FIRST(expression) = { id, num, (, not, +, - }

FOLLOW(expression) = { then, do, ), ], ,, ; }

---

19. simple_expression → term | sign term | simple_expression addop term

FIRST(simple_expression) = { id, num, (, not, +, - }

FOLLOW(simple_expression) = { relop, then, do, ), ], ,, ; }

---

20. term → factor | term mulop factor

FIRST(term) = { id, num, (, not }

FOLLOW(term) = { addop, relop, then, do, ), ], ,, ; }

---

21. factor → id  
           | id ( expression_list )  
           | num  
           | ( expression )  
           | not factor

FIRST(factor) = { id, num, (, not }

FOLLOW(factor) = { mulop, addop, relop, then, do, ), ], ,, ; }

---

22. sign → + | -

FIRST(sign) = { +, - }

FOLLOW(sign) = { id, num, (, not }
