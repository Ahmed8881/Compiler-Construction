1.  identifier_list     ->  id
                        |   identifier_list , id
 
2.  type                ->  standard_type
                        |   array [ num .. num ] of standard_type
 
3.  standard_type       ->  integer
                        |   real
 
4.  parameter_list      ->  identifier_list : type
                        |   parameter_list ; identifier_list : type
 
5.  statement_list      ->  statement
                        |   statement_list ; statement
 
6.  statement           ->  variable assignop expression
                        |   procedure_statement
                        |   if expression then statement else statement
                        |   while expression do statement
 
7.  variable            ->  id
                        |   id [ expression ]
 
8.  procedure_statement ->  id
                        |   id ( expression_list )
 
9.  expression_list     ->  expression
                        |   expression_list , expression
 
10. expression          ->  simple_expression
                        |   simple_expression relop simple_expression
 
11. simple_expression   ->  term
                        |   sign term
                        |   simple_expression addop term
 
12. term                ->  factor
                        |   term mulop factor
 
13. factor              ->  id
                        |   id ( expression_list )
                        |   num
                        |   ( expression )
                        |   not factor
 
14. sign                ->  +
                        |   -
